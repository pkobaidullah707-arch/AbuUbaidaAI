name: Build Abu Ubaida AI APK

on:
  push:
    branches: [ main, master ]
  workflow_dispatch:

jobs:
  build-debug-apk:
    runs-on: ubuntu-latest

    steps:
      - name: Checkout repository
        uses: actions/checkout@v4

      - name: Extract project ZIP
        run: |
          shopt -s nullglob
          for z in *.zip; do
            unzip -o "$z"
          done

      - name: Fix BengaliPhoneticMatcher Unicode Char Literals
        run: |
          python3 - <<'EOF'
          from pathlib import Path
          p = Path("AbuUbaidaAI/app/src/main/java/com/abuubaida/ai/phone/BengaliPhoneticMatcher.kt")
          t = p.read_text(encoding="utf-8")
          t = t.replace('"\u09ac\u09cd\u09ac" to "bb"', '"\u09ac\u09cd\u09ac" to "bb", "\\u09AF\\u09BC" to "y", "\\u09A1\\u09BC" to "r", "\\u09A2\\u09BC" to "r"')
          t = t.replace("'\u09af\u09bc'", "'\\u09DF'").replace("'\u09a1\u09bc'", "'\\u09DC'").replace("'\u09a2\u09bc'", "'\\u09DD'")
          p.write_text(t, encoding="utf-8")
          EOF

      - name: Set up JDK 17
        uses: actions/setup-java@v4
        with:
          distribution: 'temurin'
          java-version: '17'

      - name: Accept Android SDK Licenses
        run: |
          SDK_MANAGER="$(command -v sdkmanager || echo "$ANDROID_HOME/cmdline-tools/latest/bin/sdkmanager")"
          if [ -x "$SDK_MANAGER" ]; then
            yes | "$SDK_MANAGER" --licenses > /dev/null 2>&1 || true
          fi

      - name: Setup Gradle
        uses: gradle/actions/setup-gradle@v4
        with:
          gradle-version: '8.9'

      - name: Build Debug APK
        run: |
          if [ -d "AbuUbaidaAI" ]; then
            cd AbuUbaidaAI
          fi
          gradle assembleDebug --no-daemon

      - name: Upload Debug APK Artifact
        uses: actions/upload-artifact@v4
        with:
          name: AbuUbaidaAI-debug-apk
          path: "**/app/build/outputs/apk/debug/app-debug.apk"
